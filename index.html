function doPost(e) { try { const data = JSON.parse(e.postData.contents); const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("Reservas"); // Si no existe la hoja, la crea if (!sheet) { SpreadsheetApp.getActiveSpreadsheet().insertSheet("Reservas"); } // Añadir fila con los datos sheet.appendRow([ new Date(), data.nom || "", data.email || "", data.telephone || "", data.date || "", data.horaire || "" ]); return ContentService .createTextOutput(JSON.stringify({ success: true })) .setMimeType(ContentService.MimeType.JSON); } catch (error) { return ContentService .createTextOutput(JSON.stringify({ success: false, error: error.message })) .setMimeType(ContentService.MimeType.JSON); } }
